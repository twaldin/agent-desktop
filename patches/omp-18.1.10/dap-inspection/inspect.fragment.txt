export interface DapConfigInspection {
  /** Native priority, highest first. Inspection never starts an adapter. */
  sources: Array<{ path: string; scope: "project" | "user" | "plugin"; content?: string; document?: unknown; adapters?: Record<string, unknown>; ignoredAdapters?: string[]; error?: string }>;
  defaults: Record<string, DapAdapterConfig>;
  merged: Record<string, DapAdapterConfig>;
  fieldSources: Record<string, Record<string, number | "builtin">>;
  /** Nested defaults merge one level; preserve provenance for each merged key. */
  defaultSources: Record<string, { launchDefaults: Record<string, number | "builtin">; attachDefaults: Record<string, number | "builtin"> }>;
  resolved: Record<string, DapResolvedAdapter | null>;
}
/** Reuses native source order, normalization, merging and binary resolution.
 * Explicit roots avoid changing process-global plugin discovery state.
 * Replacement is a read-only preflight for a source already in the native list.
 */
export function inspectAdapterConfigs(cwd: string, roots?: readonly ClaudePluginRoot[], replacement?: { path: string; content: string }): DapConfigInspection {
  const sources: DapConfigInspection["sources"] = getConfigSources(cwd, roots).map(source => {
    const item: DapConfigInspection["sources"][number] = { path: source.path, scope: source.scope };
    try {
      try { item.content = replacement?.path === source.path ? replacement.content : readDapInspectionFile(source.path); }
      catch (error) { if ((error as NodeJS.ErrnoException).code === "ENOENT") return item; throw error; }
      item.document = parseConfigContent(item.content, source.path);
      const normalized = normalizeConfig(item.document);
      if (!normalized) throw new Error("Expected a configuration object.");
      item.adapters = normalized.adapters;
    } catch (error) { item.error = error instanceof Error ? error.message : String(error); }
    return item;
  });
  let merged = { ...DEFAULT_ADAPTERS };
  const fieldSources: DapConfigInspection["fieldSources"] = Object.fromEntries(Object.entries(merged).map(([name, config]) => [name, Object.fromEntries(Object.keys(config).map(key => [key, "builtin"]))]));
  const defaultSources: DapConfigInspection["defaultSources"] = Object.fromEntries(Object.entries(merged).map(([name, config]) => [name, {
    launchDefaults: Object.fromEntries(Object.keys(config.launchDefaults ?? {}).map(key => [key, "builtin"])),
    attachDefaults: Object.fromEntries(Object.keys(config.attachDefaults ?? {}).map(key => [key, "builtin"])),
  }]));
  for (const source of [...sources].reverse()) {
    if (source.error || !source.adapters) continue;
    const before = merged; merged = mergeAdapters(merged, source.adapters);
    source.ignoredAdapters = Object.keys(source.adapters).filter(name => !Object.hasOwn(merged, name) || merged[name] === before[name]);
    for (const [name, config] of Object.entries(merged)) {
      const raw = source.adapters[name];
      if (config === before[name] || !isRecord(raw)) continue;
      const fields = fieldSources[name] ??= {};
      const defaults = defaultSources[name] ??= { launchDefaults: {}, attachDefaults: {} };
      for (const [key, value] of Object.entries(config)) {
        if (Object.hasOwn(raw, key) || JSON.stringify(value) !== JSON.stringify((before[name] as unknown as Record<string, unknown> | undefined)?.[key])) fields[key] = sources.indexOf(source);
      }
      for (const field of ["launchDefaults", "attachDefaults"] as const) {
        if (!isRecord(raw[field])) continue;
        for (const key of Object.keys(raw[field])) defaults[field][key] = sources.indexOf(source);
      }
    }
  }
  const resolved = Object.fromEntries(Object.keys(merged).map(name => [name, resolveAdapterFromConfig(name, merged, cwd)]));
  return { sources, defaults: { ...DEFAULT_ADAPTERS }, merged, fieldSources, defaultSources, resolved };
}
function readDapInspectionFile(file: string): string {
  // Follow native-readable symlinks for accurate inspection; the host refuses writes to them.
  const fd = fs.openSync(file, fs.constants.O_RDONLY | fs.constants.O_NONBLOCK);
  try {
    const metadata = fs.fstatSync(fd);
    if (!metadata.isFile() || metadata.size > 1024 * 1024) throw new Error("DAP inspection requires a regular file at most 1 MiB; effective inspection is incomplete.");
    const bytes = Buffer.alloc(1024 * 1024 + 1); let size = 0;
    while (size < bytes.length) { const count = fs.readSync(fd, bytes, size, bytes.length - size, null); if (!count) break; size += count; }
    if (size > 1024 * 1024) throw new Error("DAP inspection exceeds 1 MiB; effective inspection is incomplete.");
    return bytes.subarray(0, size).toString("utf8");
  } finally { fs.closeSync(fd); }
}
